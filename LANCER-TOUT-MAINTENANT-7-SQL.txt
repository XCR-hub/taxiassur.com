🚀 TESTER MAINTENANT (2 minutes)

Copier-coller dans Supabase SQL:

-- 1) Lancer workflow manuellement
SELECT trigger_backlink_workflow();

-- 2) Vérifier opportunités contactées
SELECT 
  COUNT(*) FILTER (WHERE status = 'contacted') as contactes,
  COUNT(*) FILTER (WHERE status = 'new') as nouveaux
FROM backlink_opportunities;

-- 3) Vérifier emails créés
SELECT 
  COUNT(*) as nb_emails,
  status,
  template_used,
  recipient_email
FROM backlink_outreach
GROUP BY status, template_used, recipient_email
ORDER BY status;

-- 4) Détail emails pending
SELECT 
  o.domain,
  bo.recipient_email,
  bo.email_subject,
  bo.status,
  bo.created_at
FROM backlink_outreach bo
JOIN backlink_opportunities o ON bo.opportunity_id = o.id
WHERE bo.status = 'pending'
ORDER BY bo.created_at DESC;

Run ▶
