CREATE FUNCTION toggle_automation(automation_name TEXT, enabled BOOLEAN) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, cron AS $$ DECLARE affected_count INT; BEGIN UPDATE cron.job SET active = enabled WHERE jobname = automation_name; GET DIAGNOSTICS affected_count = ROW_COUNT; RETURN jsonb_build_object('success', true, 'affected_rows', affected_count); END; $$;
