```typescript
    .select('*');

async markAsRead(messageId: string) {
  const { data } = await supabase
    .from('email_replies')
    .update({ is_processed: true })
    .eq('id', messageId);  // ❌ ID avec préfixe invalide
}
```

APRÈS:
```typescript
async getInbox() {
  const { data } = await supabase
    .from('email_inbox')  // ✅ BONNE TABLE
    .select('*');
  
  console.log('📧 Emails trouvés:', data?.length);
}

async markAsRead(messageId: string) {
  const realId = messageId.replace(/^(in-|out-)/, '');  // ✅ Retrait préfixe
  
  const { data } = await supabase
    .from('email_inbox')  // ✅ BONNE TABLE
    .update({ processed: true })
    .eq('id', realId);  // ✅ UUID valide
}
```
