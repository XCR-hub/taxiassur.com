SELECT 
  CASE 
    WHEN vault.decrypted_secrets -> 'SENDGRID_API_KEY' IS NOT NULL 
    THEN '✅ SendGrid configuré'
    ELSE '❌ SendGrid MANQUANT'
  END as sendgrid_status,
  CASE 
    WHEN vault.decrypted_secrets -> 'HUNTER_IO_API_KEY' IS NOT NULL 
    THEN '✅ Hunter.io configuré'
    ELSE '❌ Hunter.io MANQUANT (optionnel)'
  END as hunter_status,
  CASE 
    WHEN vault.decrypted_secrets -> 'OPENAI_API_KEY' IS NOT NULL 
    THEN '✅ OpenAI configuré'
    ELSE '❌ OpenAI MANQUANT'
  END as openai_status;
