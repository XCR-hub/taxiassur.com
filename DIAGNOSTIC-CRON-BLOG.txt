Pour nettoyer les doublons :

-- Voir les doublons
SELECT 
  slug,
  COUNT(*) as nb_versions,
  array_agg(id ORDER BY created_at) as ids
FROM blog_posts
WHERE category = 'actualites'
GROUP BY slug
HAVING COUNT(*) > 1;

-- Supprimer les doublons (garde le plus récent)
DELETE FROM blog_posts
WHERE id IN (
  SELECT id 
  FROM (
    SELECT 
      id,
      ROW_NUMBER() OVER (PARTITION BY slug ORDER BY created_at DESC) as rn
    FROM blog_posts
    WHERE category = 'actualites'
  ) t
  WHERE rn > 1
);
